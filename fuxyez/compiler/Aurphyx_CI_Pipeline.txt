# Aurphyx Advanced CI Pipeline (for Rust Monorepo with multiple crates)
# Features:
# - Runs linting (clippy, fmt)
# - Builds all crates (core + fux + yez)
# - Runs unit and integration tests
# - Caches cargo registry and build artifacts for speed
# - Runs wasm-pack build for frontend targets (if any)
# - Publishes artifacts or Docker images on release branch

name: Aurphyx Rust Monorepo CI Pipeline

on:
  push:
    branches:
      - main
      - release/*
  pull_request:
    branches:
      - main

jobs:
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Install Rust nightly toolchain
        uses: actions-rs/toolchain@v1
        with:
          toolchain: nightly
          components: rustfmt, clippy
      - name: Run cargo fmt check
        run: cargo fmt --all -- --check
      - name: Run cargo clippy
        run: cargo clippy --all-targets --all-features -- -D warnings

  build-test:
    runs-on: ubuntu-latest
    needs: lint
    steps:
      - uses: actions/checkout@v3
      - name: Cache cargo registry and build
        uses: actions/cache@v3
        with:
          path: |
            ~/.cargo/registry
            ~/.cargo/git
            target
          key: ${{ runner.os }}-cargo-${{ hashFiles('**/Cargo.lock') }}
          restore-keys: |
            ${{ runner.os }}-cargo-
      - name: Install Rust stable toolchain
        uses: actions-rs/toolchain@v1
        with:
          toolchain: stable
      - name: Build all crates with features
        run: cargo build --all --release
      - name: Run unit and integration tests
        run: cargo test --all --release

  wasm-build:
    runs-on: ubuntu-latest
    needs: build-test
    if: contains(github.event.head_commit.message, 'wasm') || github.ref == 'refs/heads/main'
    steps:
      - uses: actions/checkout@v3
      - name: Setup wasm-pack
        run: curl https://rustwasm.github.io/wasm-pack/installer/init.sh -sSf | sh
      - name: Build wasm frontends (e.g., fux_frontend)
        working-directory: ./fux_frontend
        run: wasm-pack build --release

  release:
    needs: [lint, build-test, wasm-build]
    if: startsWith(github.ref, 'refs/heads/release/')
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Publish crates
        run: |
          cargo publish --workspace --token ${{ secrets.CARGO_TOKEN }}
      - name: Publish Docker image (if applicable)
        run: |
          docker build -t aurphyx/fuxyez:latest .
          docker push aurphyx/fuxyez:latest
